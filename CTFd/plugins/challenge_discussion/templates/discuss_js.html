// Shared JS — included inside an IIFE. // Expects: CHALLENGE_ID, POST_TYPE, CURRENT_USER_ID, IS_ADMIN, CSRF_TOKEN, ALLOW_REPLIES var replyParentId = null; var postMap = {}; // id -> post object (all levels) var MAX_INDENT = 6; // max visual indent steps var INDENT_PX = 24; // pixels per indent level function renderMarkdown(text) { if (window.marked) { var html = marked.parse(text || '', {breaks: true, gfm: true}); return window.DOMPurify ? DOMPurify.sanitize(html) : escapeHtml(text || ''); } return escapeHtml(text || ''); } function escapeHtml(text) { var d = document.createElement('div'); d.appendChild(document.createTextNode(text || '')); return d.textContent; } function appendMarkdown(el, text) { if (window.marked && window.DOMPurify) { var html = marked.parse(text || '', {breaks: true, gfm: true}); el.replaceChildren(DOMPurify.sanitize(html, {RETURN_DOM_FRAGMENT: true})); return; } el.textContent = text || ''; el.style.whiteSpace = 'pre-wrap'; } function makeEl(tag, className, text) { var el = document.createElement(tag); if (className) el.className = className; if (text !== undefined) el.textContent = text; return el; } function formatDate(iso) { return new Date(iso).toLocaleString(); } // -- Reply context bar -- function showReplyContext(post) { replyParentId = post.id; var bar = document.getElementById('reply-context'); var preview = (post.content || '').slice(0, 120) + (post.content && post.content.length > 120 ? 'transparent' : 'hsl(' + ((depth * 47) % 360) + ',60%,55%)'; var canDelete = post.user_id === CURRENT_USER_ID || IS_ADMIN; var wrapper = document.createElement('div'); wrapper.id = 'post-' + post.id; if (depth > 0) { wrapper.style.marginLeft = indent + 'px'; wrapper.style.borderLeft = '3px solid ' + borderColor; wrapper.style.paddingLeft = '10px'; } var card = makeEl('div', 'card mb-2'); var body = makeEl('div', 'card-body py-2'); var header = makeEl('div', 'd-flex justify-content-between mb-1'); header.append(makeEl('strong', '', post.username), makeEl('small', 'text-muted', formatDate(post.date))); var content = makeEl('div', 'markdown-body mt-1 mb-2'); appendMarkdown(content, post.content); var actions = document.createElement('div'); if (ALLOW_REPLIES) { var replyBtn = makeEl('button', 'btn btn-sm btn-outline-secondary reply-btn me-1'); replyBtn.dataset.postId = post.id; replyBtn.append(makeEl('i', 'fas fa-reply'), document.createTextNode(' Reply')); actions.appendChild(replyBtn); } if (canDelete) { var deleteBtn = makeEl('button', 'btn btn-sm btn-outline-danger delete-btn ms-1'); deleteBtn.dataset.postId = post.id; deleteBtn.appendChild(makeEl('i', 'fas fa-trash')); actions.appendChild(deleteBtn); } body.append(header, content, actions); card.appendChild(body); wrapper.appendChild(card); if (post.replies && post.replies.length) { post.replies.forEach(function(r) { wrapper.appendChild(buildPostElement(r, depth + 1)); }); } return wrapper; } // -- Render all top-level posts -- function renderPosts(posts) { var container = document.getElementById('discussion-posts'); if (!posts.length) { container.replaceChildren(makeEl('p', 'text-muted text-center py-3', 'No posts yet. Be the first!')); return; } postMap = {}; container.replaceChildren.apply(container, posts.map(function(p) { return buildPostElement(p, 0); })); // Bind reply buttons (all depths) if (ALLOW_REPLIES) { container.querySelectorAll('.reply-btn').forEach(function(btn) { btn.addEventListener('click', function() { var pid = parseInt(btn.dataset.postId); var post = postMap[pid]; if (post) showReplyContext(post); document.getElementById('post-form').scrollIntoView({behavior: 'smooth', block: 'center'}); }); }); } // Bind delete buttons (all depths) container.querySelectorAll('.delete-btn').forEach(function(btn) { btn.addEventListener('click', function() { if (!confirm('Delete this post?')) return; fetch('/api/v1/discussion/posts/' + btn.dataset.postId, { method: 'DELETE', headers: {'Content-Type': 'application/json', 'CSRF-Token': CSRF_TOKEN}, }).then(function(r) { return r.json(); }).then(function(data) { if (data.success) { var el = document.getElementById('post-' + btn.dataset.postId); if (el) el.remove(); } else { alert(data.errors || 'Delete failed'); } }); }); }); } function loadPosts() { fetch('/api/v1/discussion/posts?challenge_id=' + CHALLENGE_ID + '&type=' + POST_TYPE) .then(function(r) { return r.json(); }) .then(function(data) { if (data.success) { renderPosts(data.data); } else { document.getElementById('discussion-posts').replaceChildren( makeEl('div', 'alert alert-danger', data.errors) ); } }); } document.getElementById('post-form').addEventListener('submit', function(e) { e.preventDefault(); var content = document.getElementById('post-content').value.trim(); if (!content) return; var body = {challenge_id: CHALLENGE_ID, post_type: POST_TYPE, content: content}; if (replyParentId) body.parent_id = replyParentId; fetch('/api/v1/discussion/posts', { method: 'POST', headers: {'Content-Type': 'application/json', 'CSRF-Token': CSRF_TOKEN}, body: JSON.stringify(body), }).then(function(r) { return r.json(); }).then(function(data) { if (data.success) { document.getElementById('post-content').value = ''; clearReplyContext(); loadPosts(); } else { alert(data.errors); } }); }); loadPosts();
